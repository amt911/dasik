# C2 — AppArmor: Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** An `apparmor` block that installs AppArmor, makes it the active LSM through a derived kernel parameter, optionally wires up the audit daemon so denials are readable, and carries the user's own profiles — visible in `plan`, captured by `sync`.

**Architecture:** Convergence rides domains that already own themselves — the expand toggle contributes packages, units, `user_groups` and files; `KernelCmdlineAction` derives `lsm=…` (and the audit parameters) exactly as it does for `cpu`/`sysrq`. What is missing is the way *back*, so a new CAPTURE-ONLY `ApparmorAction` reconstructs the block from the machine, the same shape as `CpuAction`/`PlymouthAction`.

**Tech Stack:** Python ≥3.10, pydantic v2, pytest.

## Global Constraints

- The LSM list is `landlock,lockdown,yama,integrity,apparmor,bpf` — AppArmor must be the first *major* module, and `capability` is never listed (the kernel always includes it).
- Audit parameters: `audit=1 audit_backlog_limit=8192`.
- Explicit `kernel_cmdline` entries always win over derived ones; derived parameters are subtracted by NAME in `import_state` so they come back as the block, not as hand-set noise.
- dasik does NOT run `apparmor_parser` during an install: AppArmor is not running inside the chroot. Profiles load at the next boot.
- Desktop notifications (`aa-notify`) are out of scope — they write into `$HOME`, which is config-saver's territory (recorded as a follow-up on #173).
- TDD; coverage ≥80%; `mypy`, `bandit`, mutation gates stay clean; commit per task.

```python
"""Models for the `apparmor` block (Mandatory Access Control)."""
import re
from typing import List
from pydantic import BaseModel, Field, field_validator

# The name becomes a file under /etc/apparmor.d/, so it must stay a plain
# filename — a path separator there would write anywhere on the target.
_PROFILE_NAME_RE = re.compile(r"[A-Za-z0-9_.-]{1,128}")


class ApparmorProfile(BaseModel):
    """One profile file dropped into /etc/apparmor.d/."""
    name: str = Field(..., description="File name under /etc/apparmor.d/ (e.g. 'usr.bin.foo')")
    content: str = Field(..., description="Verbatim profile text")

    @field_validator("name")
    @classmethod
    def _validate_name(cls, v: str) -> str:
        if not _PROFILE_NAME_RE.fullmatch(v):
            raise ValueError(
                f"Invalid AppArmor profile name {v!r}: must match "
                f"[A-Za-z0-9_.-]{{1,128}} (it is a file name under /etc/apparmor.d, "
                f"not a path)."
            )
        return v


class ApparmorModel(BaseModel):
    """AppArmor. An absent block means AppArmor is not managed at all."""
    enable: bool = Field(default=True, description="Install AppArmor and make it the active LSM")
    audit: bool = Field(
        default=False,
        description="Also install the audit daemon so denials are logged and readable "
                    "(auditd + audit kernel parameters + the `audit` group)",
    )
    extra_profiles: List[ApparmorProfile] = Field(
        default_factory=list,
        description="Profiles copied verbatim into /etc/apparmor.d/",
    )
```

```python
_APPARMOR_PROFILE_DIR = "/etc/apparmor.d"
_AUDIT_TMPFILES = "/etc/tmpfiles.d/audit.conf"


def expand_apparmor(config: Dict[str, Any]) -> Dict[str, Any]:
    """AppArmor: the package, the unit, and optionally the audit framework.

    The kernel parameter that actually turns AppArmor on is derived by
    KernelCmdlineAction — without `lsm=` naming it, the package is installed and
    enforces nothing.

    The tmpfiles override is not optional decoration: Arch ships
    `z /var/log/audit 700 root root`, which systemd-tmpfiles re-applies on every
    upgrade, so a user in the `audit` group can read the log until the next
    `pacman -Syu` and never again.
    """
    cfg = config.get("apparmor")
    if cfg is None or not cfg.get("enable", True):
        return {}
    packages = ["apparmor"]
    units = ["apparmor.service"]
    files = [{"path": f"{_APPARMOR_PROFILE_DIR}/{p['name']}", "content": p["content"]}
             for p in cfg.get("extra_profiles") or []]
    out: Dict[str, Any] = {}
    if cfg.get("audit"):
        packages.append("audit")
        units.append("auditd.service")
        out["user_groups"] = ["audit"]
        files.append({"path": _AUDIT_TMPFILES,
                      "content": "# Managed by dasik\nz /var/log/audit 750 root audit - -\n"})
    out["packages"] = packages
    out["units"] = units
    if files:
        out["files"] = files
    return out
```

```python
# AppArmor must be the FIRST major module in the list, and `capability` is never
# listed — the kernel always includes it (Arch wiki, AppArmor#Installation).
_LSM_PARAM = "lsm=landlock,lockdown,yama,integrity,apparmor,bpf"
_AUDIT_PARAMS = ["audit=1", "audit_backlog_limit=8192"]
```

```python
    def _derive_from_apparmor(self) -> List[str]:
        """Kernel params for the `apparmor` block.

        Installing the package is not enough: without `lsm=` naming apparmor the
        module never initialises and every profile is inert, which looks exactly
        like a working install right up until you check `aa-enabled`.
        """
        cfg = self._cfg.get("apparmor")
        if cfg is None or not cfg.get("enable", True):
            return []
        params = [_LSM_PARAM]
        if cfg.get("audit"):
            # Denials are logged through the audit subsystem; the backlog limit
            # keeps early-boot messages from being dropped before auditd starts.
            params += list(_AUDIT_PARAMS)
        return params
```

import os
from types import SimpleNamespace

from dasik.lib.actions.apparmor_action import ApparmorAction


class _Target:
    def __init__(self, root):
        self.root = str(root)

    def path(self, canonical):
        return os.path.join(self.root, canonical.lstrip("/"))


def _machine(tmp_path, installed=True, params="root=LABEL=root rw", profiles=(),
             audit=False):
    (tmp_path / "boot/loader/entries").mkdir(parents=True, exist_ok=True)
    (tmp_path / "boot/loader/loader.conf").write_text("default arch\n")
    (tmp_path / "boot/loader/entries/arch.conf").write_text(
        f"title Arch\noptions {params}\n")
    if installed:
        (tmp_path / "usr/bin").mkdir(parents=True, exist_ok=True)
        (tmp_path / "usr/bin/apparmor_parser").write_text("")
    if audit:
        (tmp_path / "usr/bin").mkdir(parents=True, exist_ok=True)
        (tmp_path / "usr/bin/auditd").write_text("")
    profile_dir = tmp_path / "etc/apparmor.d"
    profile_dir.mkdir(parents=True, exist_ok=True)
    for name, content in profiles:
        (profile_dir / name).write_text(content)
    return tmp_path


def _action(tmp_path, cfg=None):
    return ApparmorAction({"bootloader": "sd-boot", **(cfg or {})},
                          SimpleNamespace(target=_Target(tmp_path)))


_LSM = "lsm=landlock,lockdown,yama,integrity,apparmor,bpf"
